class Solution {
public:
    int maxDistance(vector<int>& nums1, vector<int>& nums2) {
        // int n=nums1.size();
        // int m=nums2.size();
        // int mx=0;
        // for(int i=0;i<n;i++){
        //     int j=i;
        //     while(j<m && nums1[i]<=nums2[j]){
        //             mx=max(mx,j-i);
        //             j++;
        //     }
        // }
        // return mx;

        int n = nums1.size();
        int m = nums2.size();
        int i = 0, j = 0;
        int mx = 0;
        while (i < n && j < m) {
            if (nums1[i] <= nums2[j]) {
                mx = max(mx, j - i);
                j++;
            }
            else {
                i++;
            }
        }
        return mx;
    }
};
